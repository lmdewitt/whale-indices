Dygraph = require(await FileAttachment("lib/dygraph/dygraph.min.js").url())

html`${allGroups.map((group, groupIdx) => {
  
  const checkboxInputs = Inputs.checkbox(
    group.titles, 
    { 
      value: group.titles.filter((_, i) => group.visibility[i])
    }
  );
  
  const indinfo = Inputs.checkbox(
    group.titles, 
    { 
      value: group.titles.filter((_, i) => group.visibility[i])
    }
  );
  
  const legdivht = 50 + (group.titles.length*30);
  
  
  // Generate dynamic CSS rules to style each checkbox label & input by color
  const dynamicStyles = html`<style>
    ${group.colors.map((color, idx) => `
      /* Scope styles specifically to this group index */
      .group-cb-${groupIdx} label:nth-child(${idx + 1}) {
        color: ${color} !important;
        font-weight: 400;
        display: block;
      }
      .group-cb-${groupIdx} label:nth-child(${idx + 1}) input[type="checkbox"] {
        accent-color: ${color} !important;
        color: white;
      }
    `).join("")}
    #legdiv_${groupIdx} {height : ${legdivht} px;}
  </style>`;

  const chartContainer = document.createElement("div");
  chartContainer.style.width = "100%";
  chartContainer.style.height = "300px";
  chartContainer.style.position = "relative";
  chartContainer.style.marginTop = "10px";
  chartContainer.style.marginBottom = "40px";

  const chartData = group.timestamps.map((t, rowIdx) => {
    const row = [new Date(t)];
    for (let colIdx = 0; colIdx < group.labels.length; colIdx++) {
      row.push(group.values[rowIdx][colIdx]);
    }
    return row;
  });

  checkboxInputs.addEventListener("input", () => {
    const selectedTitles = checkboxInputs.value;
    const currentVisibility = group.titles.map(title => selectedTitles.includes(title));
  
    if (chartContainer._dygraphInstance) {
      chartContainer._dygraphInstance.updateOptions({ visibility: currentVisibility });
    }
  });

  requestAnimationFrame(() => {
    const initialVisibility = group.titles.map(title => checkboxInputs.value.includes(title));
    
    chartContainer._dygraphInstance = new Dygraph(chartContainer, chartData, {
      labels: ["Time", ...group.labels],
      colors: group.colors,
      visibility: initialVisibility,
      ylabel: group.ylabel || "",
      y2label: group.y2label || "",
      series: group.series || {},  // Assigns specified series to 'y2' axis
      legend: "always",
      hideOverlayOnMouseOut: false,
      showRangeSelector: true,
      labelsSeparateLines: true,
      labelsDiv: "legdiv_"+group.index,
      strokeWidth: group.strokewidth,
      pointSize: group.pointsize,
      highlightCircleSize: group.highlightcirclesize,
      dateWindow: group.datewindow,
      xRangePad: group.xrangepad,
      yRangePad: group.yrangepad,
      rightGap: group.rightgap
    });

    setTimeout(() => {
      chartContainer._dygraphInstance.resize();
    }, 50);
  });

  return html`
    ${dynamicStyles}
    <div id="grp${groupIdx}" class="custom-container" style="padding: 15px; background: #ffffff; border: 1px solid #e9ecef; border-radius: 6px; margin-bottom: 25px;">
      <div class="custom-sidebar group-cb-${groupIdx}">
      <span class="custom-sidebar-title">Choose indicator:</span>
      ${checkboxInputs}
      </div>
      <div class="custom-main">
      <div><h3>${group.title}</h3></div>
      <div id="legdiv_${group.index}"></div>
      ${chartContainer}
      </div>
    </div>
  `;
})}`
